# EconRadar backend — runtime dependencies.
# Python 3.12 target (3.13 works locally). The heavy forecasting weights
# (Chronos-2, TimesFM) are NOT here: they live in the Modal image, never on the
# 1 vCPU / 2 GB VPS — architecture decisions #21 and #22.

fastapi>=0.115,<0.116
uvicorn[standard]>=0.30,<0.35
pydantic>=2.7,<3
pydantic-settings>=2.3,<3

# Database — psycopg3 is the single Postgres driver: SQLAlchemy async engine
# (postgresql+psycopg) for the API AND APScheduler's sync SQLAlchemyJobStore.
SQLAlchemy>=2.0.30,<2.1
psycopg[binary]>=3.2,<3.3

# HTTP client for data-source connectors.
httpx>=0.27,<0.29

# Scheduler — 3.x per architecture.md (AsyncIOScheduler + SQLAlchemyJobStore).
APScheduler>=3.10,<4.0

# Modal — external inference host for Chronos-2 (architecture decision #21).
# Listed as a runtime dependency, not a Phase 3 "heavy AI/ML dep", because:
#   * it is a thin gRPC client (~22 small packages, no torch/transformers), so the
#     "keep the free-tier build small" rule above does not apply;
#   * it is already installed in the production venv and the service unit already
#     loads MODAL_TOKEN_ID / MODAL_TOKEN_SECRET, so omitting it would make this file
#     a false record of the runtime — and a venv rebuild would silently drop it;
#   * deploying the Modal function needs the CLI regardless of whether the app later
#     calls it via the SDK or over HTTP, so it is required either way.
# The Chronos-2 weights themselves live in the Modal image, never on this box.
modal>=1.5,<2

# pgvector's SQLAlchemy types — maps the existing embeddings.embedding vector(384)
# column (feature 2.2). Pure Python, no compiled extension. A hard import in
# models.py, so it is a runtime dependency rather than a lazy one.
pgvector>=0.3,<1

# Nixtla StatsForecast — the last step of the forecasting cascade, and the ONLY one
# that runs on this box (decision #22). Imported lazily inside the fallback path, so
# numba/numpy cost nothing until Modal is actually unreachable.
statsforecast>=2.0,<3

# Jinja2 — LLM prompt templates (feature 1.5). Already present transitively via
# FastAPI's ecosystem, declared here because prompts depend on it directly.
jinja2>=3.1,<4

# fastembed — runs sentence-transformers/all-MiniLM-L6-v2 (384-d, the width the
# embeddings table was created at) through onnxruntime instead of torch. Decision
# #23: local-first embeddings mean retrieval has no external dependency and no
# shared rate limit, and ONNX keeps ~1 GB of torch off a 2 GB box. Loaded lazily,
# so a deployment that never serves a chat query never pays for it.
fastembed>=0.4,<1

# Matplotlib (Agg backend) — server-side chart PNGs for the VLM pipeline
# (feature 2.1). NOT Plotly + Kaleido: Kaleido 1.x drives an external headless
# Chrome per render, which this 1 vCPU / 2 GB box cannot afford alongside the API
# and five ingestion jobs, and 0.2.1's bundled Chromium is unmaintained. Decision
# #25 — decision #9's substance (deterministic server-side rendering) is unchanged.
matplotlib>=3.9,<4
